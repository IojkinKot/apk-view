cmake_minimum_required(VERSION 3.24)
project(apkview LANGUAGES C CXX)
if(NOT MSVC)
  message(FATAL_ERROR "Build with the Microsoft C++ toolchain.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(apk_windows STATIC src/windows/archive_path.cpp src/windows/output_file.cpp)
target_include_directories(apk_windows PUBLIC src)
target_compile_options(apk_windows PRIVATE /W4 /utf-8)
if(BUILD_TESTING)
  add_executable(windows_tests tests/windows_tests.cpp)
  target_compile_options(windows_tests PRIVATE /W4 /utf-8)
  target_link_libraries(windows_tests PRIVATE apk_windows)
  add_test(NAME windows_tests COMMAND windows_tests)
endif()
option(APKVIEW_NO_ZLIB "Development build: only uncompressed ADB support, no release ZIP" OFF)
if(NOT APKVIEW_NO_ZLIB)
  include(FetchContent)
  set(APKVIEW_ZLIB_ARCHIVE "https://zlib.net/zlib-1.4.tar.gz" CACHE STRING "URL or local path to the zlib 1.4 source archive")
  # Native Windows paths are otherwise embedded as escape sequences in the sub-build.
  set(zlib_archive_url "${APKVIEW_ZLIB_ARCHIVE}")
  if(NOT zlib_archive_url MATCHES "^[A-Za-z][A-Za-z0-9+.-]*://")
    file(TO_CMAKE_PATH "${zlib_archive_url}" zlib_archive_url)
  endif()
  # Verify and unpack the release, but do not run its build scripts.
  FetchContent_Declare(zlib URL "${zlib_archive_url}"
    URL_HASH SHA256=02dd159783677a749c04adbdd02b55b8a2bcc7e054f51b76b549199a916ad584
    SOURCE_SUBDIR apkview-no-upstream-build)
  FetchContent_MakeAvailable(zlib)
  set(zlib_files adler32.c compress.c crc32.c deflate.c gzclose.c gzlib.c gzread.c gzwrite.c
    infback.c inffast.c inflate.c inftrees.c trees.c uncompr.c zutil.c)
  list(TRANSFORM zlib_files PREPEND "${zlib_SOURCE_DIR}/")
  add_library(apk_zlib STATIC ${zlib_files})
  target_include_directories(apk_zlib SYSTEM PUBLIC "${zlib_SOURCE_DIR}")
  target_compile_definitions(apk_zlib PRIVATE ZLIB_BUILD NO_FSEEKO _CRT_SECURE_NO_DEPRECATE _CRT_NONSTDC_NO_DEPRECATE)
  add_library(ZLIB::ZLIBSTATIC ALIAS apk_zlib)
endif()
add_library(apk_core STATIC src/apk/package.cpp src/apk/stream.cpp)
target_include_directories(apk_core PUBLIC src)
target_compile_options(apk_core PRIVATE /W4 /utf-8)
if(NOT APKVIEW_NO_ZLIB)
  target_compile_definitions(apk_core PUBLIC APKVIEW_HAVE_ZLIB)
  target_link_libraries(apk_core PUBLIC ZLIB::ZLIBSTATIC)
endif()
if(BUILD_TESTING)
  add_executable(apk_tests tests/apk_tests.cpp)
  target_compile_options(apk_tests PRIVATE /W4 /utf-8)
  target_link_libraries(apk_tests PRIVATE apk_core)
  add_test(NAME apk_tests COMMAND apk_tests)
endif()

add_library(apkview SHARED src/wcx/plugin.cpp)
target_compile_options(apkview PRIVATE /W4 /utf-8)
target_link_libraries(apkview PRIVATE apk_core apk_windows bcrypt)
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
  set(wcx_arch x64)
  set_target_properties(apkview PROPERTIES SUFFIX ".wcx64")
else()
  set(wcx_arch x86)
  set_target_properties(apkview PROPERTIES SUFFIX ".wcx")
endif()
target_sources(apkview PRIVATE src/wcx/exports-${wcx_arch}.def)
if(BUILD_TESTING)
  add_executable(wcx_tests tests/wcx_tests.cpp)
  target_include_directories(wcx_tests PRIVATE src)
  target_compile_options(wcx_tests PRIVATE /W4 /utf-8)
  target_link_libraries(wcx_tests PRIVATE apk_core)
  add_test(NAME wcx_tests COMMAND wcx_tests "$<TARGET_FILE:apkview>")
  set(APKVIEW_REAL_PACKAGE "" CACHE FILEPATH "Path to the official OpenWrt 25.12.0 zlib APK for integration testing")
  if(APKVIEW_REAL_PACKAGE AND NOT APKVIEW_NO_ZLIB)
    file(SHA256 "${APKVIEW_REAL_PACKAGE}" sample_sha256)
    if(NOT sample_sha256 STREQUAL "2e72549472ec1c1aee38a149f5757aa0a216a6978bf3cd567312538cdee78d5f")
      message(FATAL_ERROR "APKVIEW_REAL_PACKAGE does not match the expected official fixture.")
    endif()
    add_test(NAME real_openwrt_package COMMAND wcx_tests "$<TARGET_FILE:apkview>" "${APKVIEW_REAL_PACKAGE}")
  endif()
endif()

if(NOT APKVIEW_NO_ZLIB)
  install(TARGETS apkview RUNTIME DESTINATION .)
  install(FILES README.md packaging/pluginst.inf DESTINATION .)
  install(FILES "${zlib_SOURCE_DIR}/LICENSE" DESTINATION . RENAME LICENSE-zlib.txt)
endif()
